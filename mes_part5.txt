async function ensureXlsxLib(){
  if(window.XLSX)return;
  await new Promise((resolve,reject)=>{const sc=document.createElement('script');sc.src='https://cdn.jsdelivr.net/npm/xlsx@0.18.5/dist/xlsx.full.min.js';sc.onload=resolve;sc.onerror=reject;document.head.appendChild(sc)});
}
function pickInvValue(row,names){
  const keys=Object.keys(row||{}); const norm=s=>String(s||'').replace(/\s/g,'').toLowerCase();
  for(const name of names){const k=keys.find(k=>norm(k)===norm(name)||norm(k).includes(norm(name)));if(k!=null&&row[k]!==''&&row[k]!=null)return row[k]}
  return '';
}
async function loadBizmekaInventory(input){
  const f=input.files&&input.files[0]; if(!f)return;
  const info=document.getElementById('bizInvInfo'); info.textContent='비즈메카 재고 파일 읽는 중...';
  try{
    await ensureXlsxLib();
    const data=await f.arrayBuffer(); const wb=XLSX.read(data,{type:'array'}); const ws=wb.Sheets[wb.SheetNames[0]];
    const rows=XLSX.utils.sheet_to_json(ws,{defval:''});
    const mapped=rows.map(r=>{
      const code=pickInvValue(r,['품목코드','품목번호','자재코드','Item Code','Code']);
      const name=pickInvValue(r,['품목명','자재명','제품명','Item Name','Name']);
      const stock=Number(String(pickInvValue(r,['현재고','재고수량','현재재고','재고','Stock','On Hand'])).replace(/,/g,''))||0;
      const reserved=Number(String(pickInvValue(r,['예약','예약수량','할당수량','Reserved'])).replace(/,/g,''))||0;
      const incoming=String(pickInvValue(r,['수입검사','검사상태','수입검사상태','Inspection'])||'PASS').trim();
      return [String(code).trim(),String(name).trim(),stock,reserved,incoming];
    }).filter(x=>x[0]||x[1]);
    if(!mapped.length)throw new Error('품목코드/품목명 열을 찾지 못했습니다.');
    s.inventory=mapped; save(); info.textContent=`불러오기 완료 / ${f.name} / ${mapped.length}개 품목`;
  }catch(e){info.textContent='불러오기 실패 / '+(e.message||e); alert('비즈메카 재고 파일을 확인해 주세요.\n품목코드, 품목명, 현재고(또는 재고수량) 열이 필요합니다.');}
  finally{input.value='';}
}
